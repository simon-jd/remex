cmake_minimum_required(VERSION 3.20)
project(remex_worker LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
# Optional project-local development headers. Production builds use normal packages.
set(REMEX_DEPS_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../.deps/worker-prefix/usr" CACHE PATH "Locally unpacked development dependencies")
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "Matched protoc" FORCE)
find_package(Protobuf REQUIRED)
find_path(GRPC_INCLUDE grpcpp/grpcpp.h HINTS "${REMEX_DEPS_PREFIX}/include" REQUIRED)
find_path(OPENSSL_INCLUDE openssl/evp.h HINTS "${REMEX_DEPS_PREFIX}/include" REQUIRED)
find_library(GRPCPP_LIBRARY NAMES grpc++ libgrpc++.so.1.51 REQUIRED)
find_library(GRPC_LIBRARY NAMES grpc libgrpc.so.29 REQUIRED)
find_library(GPR_LIBRARY NAMES gpr libgpr.so.29 REQUIRED)
find_library(CRYPTO_LIBRARY NAMES crypto libcrypto.so.3 REQUIRED)
find_library(ABSL_SYNC_LIBRARY NAMES absl_synchronization libabsl_synchronization.so.20260107 libabsl_synchronization.so.20220623 REQUIRED)
find_program(GRPC_CPP_PLUGIN grpc_cpp_plugin HINTS "${REMEX_DEPS_PREFIX}/bin" REQUIRED)
# Match protoc to the installed libprotobuf, ignoring unrelated protoc from PATH.
find_program(REMEX_PROTOC protoc HINTS /usr/bin NO_DEFAULT_PATH REQUIRED)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
file(GLOB_RECURSE THIRD_PARTY_PROTOS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/third_party" "${CMAKE_CURRENT_SOURCE_DIR}/third_party/*.proto")
set(PROTO_SOURCES)
foreach(PROTO IN LISTS THIRD_PARTY_PROTOS)
  string(REPLACE ".proto" ".pb.cc" SOURCE "${PROTO}")
  string(REPLACE ".proto" ".pb.h" HEADER "${PROTO}")
  add_custom_command(OUTPUT "${GENERATED}/${SOURCE}" "${GENERATED}/${HEADER}"
    COMMAND "${REMEX_PROTOC}" --cpp_out=${GENERATED} -I${CMAKE_CURRENT_SOURCE_DIR}/third_party -I${Protobuf_INCLUDE_DIRS} "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${PROTO}"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${PROTO}" VERBATIM)
  list(APPEND PROTO_SOURCES "${GENERATED}/${SOURCE}")
endforeach()
set(WORKER_PROTO "${CMAKE_CURRENT_SOURCE_DIR}/../api/worker/v1/worker.proto")
add_custom_command(OUTPUT "${GENERATED}/api/worker/v1/worker.pb.cc" "${GENERATED}/api/worker/v1/worker.pb.h"
  COMMAND "${REMEX_PROTOC}" --cpp_out=${GENERATED} -I${CMAKE_CURRENT_SOURCE_DIR}/.. -I${Protobuf_INCLUDE_DIRS} "${WORKER_PROTO}"
  DEPENDS "${WORKER_PROTO}" VERBATIM)
list(APPEND PROTO_SOURCES "${GENERATED}/api/worker/v1/worker.pb.cc")
foreach(PROTO build/bazel/remote/execution/v2/remote_execution.proto google/bytestream/bytestream.proto api/worker/v1/worker.proto)
  string(REPLACE ".proto" ".grpc.pb.cc" SOURCE "${PROTO}")
  string(REPLACE ".proto" ".grpc.pb.h" HEADER "${PROTO}")
  if(PROTO MATCHES "^api/")
    set(INPUT "${CMAKE_CURRENT_SOURCE_DIR}/../${PROTO}")
  else()
    set(INPUT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${PROTO}")
  endif()
  add_custom_command(OUTPUT "${GENERATED}/${SOURCE}" "${GENERATED}/${HEADER}"
    COMMAND "${REMEX_PROTOC}" --grpc_out=${GENERATED} --plugin=protoc-gen-grpc=${GRPC_CPP_PLUGIN} -I${CMAKE_CURRENT_SOURCE_DIR}/third_party -I${CMAKE_CURRENT_SOURCE_DIR}/.. -I${Protobuf_INCLUDE_DIRS} "${INPUT}"
    DEPENDS "${INPUT}" VERBATIM)
  list(APPEND PROTO_SOURCES "${GENERATED}/${SOURCE}")
endforeach()
add_library(remex_proto STATIC ${PROTO_SOURCES})
target_include_directories(remex_proto PUBLIC "${GENERATED}" "${GRPC_INCLUDE}")
target_link_libraries(remex_proto PUBLIC protobuf::libprotobuf "${GRPCPP_LIBRARY}" "${GRPC_LIBRARY}" "${GPR_LIBRARY}" "${ABSL_SYNC_LIBRARY}")
add_library(remex_core STATIC src/core.cc src/process.cc src/nvml.cc src/remote.cc src/polling.cc)
target_include_directories(remex_core PUBLIC include "${OPENSSL_INCLUDE}" "${REMEX_DEPS_PREFIX}/include/${CMAKE_LIBRARY_ARCHITECTURE}")
target_link_libraries(remex_core PUBLIC remex_proto "${CRYPTO_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(remex_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(remex-worker src/main.cc)
target_link_libraries(remex-worker PRIVATE remex_core)
add_executable(worker-tests tests/core_test.cc)
target_link_libraries(worker-tests PRIVATE remex_core)
add_test(NAME worker-core COMMAND worker-tests)
add_executable(cgroup-tests tests/cgroup_test.cc)
target_link_libraries(cgroup-tests PRIVATE remex_core)
add_test(NAME worker-cgroup COMMAND cgroup-tests)
set_tests_properties(worker-cgroup PROPERTIES SKIP_RETURN_CODE 77)
add_executable(remote-tests tests/remote_test.cc)
target_link_libraries(remote-tests PRIVATE remex_core)
add_test(NAME worker-remote COMMAND remote-tests)

add_executable(polling-tests tests/polling_test.cc)
target_link_libraries(polling-tests PRIVATE remex_core)
add_test(NAME worker-polling COMMAND polling-tests)

add_executable(nvml-tests tests/nvml_test.cc)
target_link_libraries(nvml-tests PRIVATE remex_core)
add_test(NAME worker-nvml COMMAND nvml-tests)
set_tests_properties(worker-nvml PROPERTIES SKIP_RETURN_CODE 77)
